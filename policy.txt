<script>
    // Simple toggle functionality for concept cards - NO auto-progress, just expand/collapse
    (function() {
        // Setup click toggles for concept cards
        function setupConceptCards() {
            const cards = document.querySelectorAll('.concept-card');
            cards.forEach(card => {
                const summaryDiv = card.querySelector('.concept-summary');
                if (summaryDiv) {
                    summaryDiv.addEventListener('click', (e) => {
                        // Don't toggle if clicking on the learn-more button or its children
                        if (e.target.closest('.learn-more-btn')) return;
                        // Toggle active class
                        card.classList.toggle('active');
                    });
                }
            });
        }

        // Optional: notify when clicking on learn-more buttons (just to confirm navigation)
        function setupLearnMoreButtons() {
            const buttons = document.querySelectorAll('.learn-more-btn');
            buttons.forEach(btn => {
                btn.addEventListener('click', (e) => {
                    // Let the link work normally - no extra alerts
                    console.log('Navigating to:', btn.getAttribute('href'));
                });
            });
        }

        // Initialize on page load
        document.addEventListener('DOMContentLoaded', () => {
            setupConceptCards();
            setupLearnMoreButtons();
            
            // Optional: add a small notification for future expansion placeholder click
            const chips = document.querySelectorAll('.placeholder-chip');
            chips.forEach(chip => {
                chip.style.cursor = 'pointer';
                chip.addEventListener('click', () => {
                    const notification = document.createElement('div');
                    notification.className = 'notification';
                    notification.innerHTML = `<i class="fas fa-seedling"></i> Coming soon: In-depth content for ${chip.innerText.trim()}! This space is reserved for future expansion.`;
                    document.body.appendChild(notification);
                    setTimeout(() => { notification.remove(); }, 3000);
                });
            });
        });
    })();
</script>
