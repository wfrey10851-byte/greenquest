    <script>
        // Simple Progress System
        function getProgress() {
            try {
                const saved = localStorage.getItem('greenquest_progress');
                if (saved) {
                    return JSON.parse(saved);
                }
            } catch (e) {
                console.error('Error loading progress:', e);
            }
            
            // Default progress
            return {
                ecology: 0,
                climate: 0,
                living: 0,
                waste: 0,
                trade: 0,
                policy: 0,
                totalPoints: 0,
                streak: 0,
                activities: []
            };
        }

        function updateProgressDisplay() {
            const progress = getProgress();
            
            // Calculate overall progress
            const topics = ['ecology', 'climate', 'living', 'waste', 'trade', 'policy'];
            let totalProgress = 0;
            let completedLessons = 0;
            
            topics.forEach(topic => {
                const topicProgress = progress[topic] || 0;
                totalProgress += topicProgress;
                
                // Update individual progress bars
                const progressBar = document.getElementById(`${topic}Progress`);
                const percentText = document.getElementById(`${topic}Percent`);
                
                if (progressBar) {
                    progressBar.style.width = `${topicProgress}%`;
                }
                if (percentText) {
                    percentText.textContent = `${topicProgress}%`;
                }
                
                // Count lessons (assuming each 20% = 1 lesson)
                completedLessons += Math.floor(topicProgress / 20);
            });
            
            // Update stats
            const avgProgress = Math.round(totalProgress / topics.length);
            document.getElementById('totalProgress').textContent = `${avgProgress}%`;
            document.getElementById('lessonsCompleted').textContent = completedLessons;
            document.getElementById('knowledgePoints').textContent = progress.totalPoints || 0;
            document.getElementById('currentStreak').textContent = progress.streak || 0;
            
            // Update activity list
            updateActivityList(progress.activities || []);
        }

        function updateActivityList(activities) {
            const activityList = document.getElementById('activityList');
            if (!activityList) return;
            
            // Clear existing activities (except the welcome message)
            while (activityList.children.length > 1) {
                activityList.removeChild(activityList.lastChild);
            }
            
            // Add recent activities
            const recentActivities = activities.slice(-5).reverse();
            
            recentActivities.forEach(activity => {
                const activityItem = document.createElement('div');
                activityItem.className = 'activity-item';
                
                const iconMap = {
                    'ecology': '🌿',
                    'climate': '⚖️',
                    'living': '🌞',
                    'waste': '🗑️',
                    'trade': '🤝',
                    'policy': '📈'
                };
                
                const colorMap = {
                    'ecology': 'var(--ecology-color)',
                    'climate': 'var(--climate-color)',
                    'living': 'var(--living-color)',
                    'waste': 'var(--waste-color)',
                    'trade': 'var(--trade-color)',
                    'policy': 'var(--policy-color)'
                };
                
                const icon = iconMap[activity.topic] || '📚';
                const color = colorMap[activity.topic] || 'var(--primary-color)';
                
                activityItem.innerHTML = `
                    <div class="activity-icon" style="background: ${color}">${icon}</div>
                    <div class="activity-details">
                        <p>${activity.message}</p>
                        <p class="activity-time">${activity.time}</p>
                    </div>
                `;
                
                activityList.appendChild(activityItem);
            });
        }

        function viewModuleDetails(module) {
            const progress = getProgress();
            const moduleProgress = progress[module] || 0;
            const moduleNames = {
                ecology: 'Ecology & Ecosystems',
                climate: 'Climate Justice',
                living: 'Sustainable Living',
                waste: 'Waste Management',
                trade: 'Fair Trade',
                policy: 'Policy & Economy'
            };
            
            alert(`${moduleNames[module]}\nProgress: ${moduleProgress}%\n\nContinue learning to increase your progress!`);
        }

        // Initialize on page load
        document.addEventListener('DOMContentLoaded', function() {
            updateProgressDisplay();
            
            // Add hover effects
            document.querySelectorAll('.module-card, .action-card').forEach(card => {
                card.addEventListener('mouseenter', function() {
                    this.style.transform = 'translateY(-5px)';
                });
                card.addEventListener('mouseleave', function() {
                    this.style.transform = 'translateY(0)';
                });
            });
        });

        // Listen for storage updates from other pages
        window.addEventListener('storage', function(e) {
            if (e.key === 'greenquest_progress') {
                updateProgressDisplay();
            }
        });
    </script>
