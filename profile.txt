        class AuthSystem {
            constructor() {
                console.log('AuthSystem initialized');
                this.init();

            init() {
                this.checkLoginStatus();
                this.setupEventListeners();
            }
            
            checkLoginStatus() {
                console.log('Checking login status...');
                const user = this.getCurrentUser();
                const loginBtn = document.getElementById('loginBtn');
                const userMenu = document.getElementById('userMenu');
                
                console.log('User found:', user);
                
                if (user) {
                    // User is logged in
                    console.log('User is logged in:', user.name);
                    if (loginBtn) loginBtn.style.display = 'none';
                    if (userMenu) {
                        userMenu.style.display = 'block';
                        const initial = user.name ? user.name.charAt(0).toUpperCase() : 'U';
                        document.getElementById('userAvatar').textContent = initial;
                    }
                } else {
                    // User is not logged in
                    console.log('No user logged in');
                    if (loginBtn) loginBtn.style.display = 'block';
                    if (userMenu) userMenu.style.display = 'none';
                    
                    // Show login modal if on profile page
                    if (window.location.pathname.includes('profile.html')) {
                        console.log('On profile page without login, showing modal');
                        this.showLoginModal();
                    }
                }
            }
            
            getCurrentUser() {
                try {
                    const user = localStorage.getItem('greenquest_user');
                    console.log('Raw user from localStorage:', user);
                    if (!user) {
                        console.log('No user data in localStorage');
                        return null;
                    }
                    const parsed = JSON.parse(user);
                    console.log('Parsed user:', parsed);
                    return parsed;
                } catch (e) {
                    console.error('Error parsing user:', e);
                    return null;
                }
            }
            
            getAllUsers() {
                try {
                    const users = localStorage.getItem('greenquest_users');
                    if (!users) {
                        console.log('No users in localStorage, returning empty array');
                        return [];
                    }
                    return JSON.parse(users);
                } catch (e) {
                    console.error('Error parsing users:', e);
                    return [];
                }
            }
            
            saveUsers(users) {
                localStorage.setItem('greenquest_users', JSON.stringify(users));
                console.log('Users saved:', users);
            }
            
            saveCurrentUser(user) {
                localStorage.setItem('greenquest_user', JSON.stringify(user));
                console.log('Current user saved:', user);
                this.checkLoginStatus();
            }
            
            logout() {
                console.log('Logging out...');
                localStorage.removeItem('greenquest_user');
                this.showNotification('Logged out successfully', 'success');
                setTimeout(() => {
                    window.location.href = 'home.html';
                }, 1000);
            }
            
            setupEventListeners() {
                console.log('Setting up event listeners...');
                
                // Login button
                const loginBtn = document.getElementById('loginBtn');
                if (loginBtn) {
                    loginBtn.addEventListener('click', (e) => {
                        e.preventDefault();
                        console.log('Login button clicked');
                        this.showLoginModal();
                    });
                }
                
                // Close buttons
                const closeLogin = document.getElementById('closeLogin');
                const closeRegister = document.getElementById('closeRegister');
                const closeEditProfile = document.getElementById('closeEditProfile');
                
                if (closeLogin) {
                    closeLogin.addEventListener('click', () => {
                        console.log('Close login clicked');
                        this.hideLoginModal();
                    });
                }
                
                if (closeRegister) {
                    closeRegister.addEventListener('click', () => {
                        console.log('Close register clicked');
                        this.hideRegisterModal();
                    });
                }
                
                if (closeEditProfile) {
                    closeEditProfile.addEventListener('click', () => {
                        console.log('Close edit profile clicked');
                        this.hideEditProfileModal();
                    });
                }
                
                // Switch between login and register
                const showRegister = document.getElementById('showRegister');
                const showLogin = document.getElementById('showLogin');
                
                if (showRegister) {
                    showRegister.addEventListener('click', (e) => {
                        e.preventDefault();
                        console.log('Show register clicked');
                        this.hideLoginModal();
                        this.showRegisterModal();
                    });
                }
                
                if (showLogin) {
                    showLogin.addEventListener('click', (e) => {
                        e.preventDefault();
                        console.log('Show login clicked');
                        this.hideRegisterModal();
                        this.showLoginModal();
                    });
                }
                
                // Logout button
                const logoutBtn = document.getElementById('logoutBtn');
                if (logoutBtn) {
                    logoutBtn.addEventListener('click', (e) => {
                        e.preventDefault();
                        console.log('Logout button clicked');
                        this.logout();
                    });
                }
                
                // Login form submission
                const loginForm = document.getElementById('loginForm');
                if (loginForm) {
                    loginForm.addEventListener('submit', (e) => {
                        e.preventDefault();
                        console.log('Login form submitted');
                        this.handleLogin();
                    });
                }
                
                // Register form submission
                const registerForm = document.getElementById('registerForm');
                if (registerForm) {
                    registerForm.addEventListener('submit', (e) => {
                        e.preventDefault();
                        console.log('Register form submitted');
                        this.handleRegister();
                    });
                }

                const editProfileForm = document.getElementById('editProfileForm');
                if (editProfileForm) {
                    editProfileForm.addEventListener('submit', (e) => {

                        console.log('Edit profile form submitted');
                        this.handleEditProfile();

                window.addEventListener('click', (e) => {
                    if (e.target.id === 'loginModal') {
                        this.hideLoginModal();
                    }
                    if (e.target.id === 'registerModal') {
                        this.hideRegisterModal();
                    }
                    if (e.target.id === 'editProfileModal') {
                        this.hideEditProfileModal();
                    }
                });
            }

            handleLogin() {
                console.log('Handling login...');
                const email = document.getElementById('loginEmail').value.trim();
                const password = document.getElementById('loginPassword').value;
                
                console.log('Login attempt:', { email, passwordLength: password.length });
                
                if (!email || !password) {
                    this.showNotification('Please fill in all fields', 'error');
                    return;
                }
                
                const users = this.getAllUsers();
                console.log('All users:', users);
                
                const user = users.find(u => u.email === email && u.password === password);
                
                if (user) {
                    console.log('Login successful:', user);
                    this.saveCurrentUser(user);
                    this.hideLoginModal();
                    this.showNotification('Welcome back to GreenQuest!', 'success');
                    
                    // Reset form
                    document.getElementById('loginForm').reset();
                    
                    // Reload page
                    setTimeout(() => {
                        window.location.reload();

                } else {
                    console.log('Login failed: invalid credentials');
                    this.showNotification('Invalid email or password', 'error');
                }
            }
            
            handleRegister() {
                console.log('Handling registration...');
                const name = document.getElementById('registerName').value.trim();
                const email = document.getElementById('registerEmail').value.trim();
                const password = document.getElementById('registerPassword').value;
                const confirm = document.getElementById('registerConfirm').value;
                
                console.log('Registration attempt:', { name, email, passwordLength: password.length });
                
                // Validation
                if (!name || !email || !password || !confirm) {
                    this.showNotification('Please fill in all fields', 'error');
                    return;
                }
                
                if (password !== confirm) {
                    this.showNotification('Passwords do not match', 'error');
                    return;
                }
                
                if (password.length < 6) {
                    this.showNotification('Password must be at least 6 characters', 'error');
                    return;
                }
                
                // Check if user already exists
                const users = this.getAllUsers();
                if (users.some(u => u.email === email)) {
                    this.showNotification('Email already registered', 'error');
                    return;
                }
                
                // Create new user
                const newUser = {
                    id: Date.now(),
                    name: name,
                    email: email,
                    password: password,
                    joinDate: new Date().toISOString(),
                    bio: '',
                    level: 1,
                    points: 0,
                    completedQuests: 0,
                    streakDays: 1,
                    carbonSaved: 0
                };
                
                console.log('New user created:', newUser);
                
                // Save user
                users.push(newUser);
                this.saveUsers(users);
                this.saveCurrentUser(newUser);
                
                // Initialize user progress
                this.initializeUserProgress();
                
                this.hideRegisterModal();
                this.showNotification('Welcome to GreenQuest! Your account has been created.', 'success');
                
                // Reset form
                document.getElementById('registerForm').reset();
                
                // Reload page
                setTimeout(() => {
                    window.location.reload();
                }, 1500);
            }
            
            handleEditProfile() {
                console.log('Handling edit profile...');
                const name = document.getElementById('editName').value.trim();
                const email = document.getElementById('editEmail').value.trim();
                const bio = document.getElementById('editBio').value.trim();
                
                if (!name || !email) {
                    this.showNotification('Name and email are required', 'error');
                    return;
                }
                
                const user = this.getCurrentUser();
                const users = this.getAllUsers();
                
                // Check if email is already taken by another user
                if (email !== user.email && users.some(u => u.email === email)) {
                    this.showNotification('Email already taken by another user', 'error');
                    return;
                }
                
                // Update user
                user.name = name;
                user.email = email;
                if (bio) user.bio = bio;
                
                console.log('Updated user:', user);
                
                // Update in users array
                const userIndex = users.findIndex(u => u.id === user.id);
                if (userIndex !== -1) {
                    users[userIndex] = user;
                    this.saveUsers(users);
                }
                
                this.saveCurrentUser(user);
                this.hideEditProfileModal();
                this.showNotification('Profile updated successfully', 'success');
                
                // Update display
                setTimeout(() => {
                    window.location.reload();
                }, 1000);
            }

            showNotification(message, type = 'info') {
                console.log('Showing notification:', { message, type });
                
                // Remove existing notification
                const existing = document.querySelector('.notification');
                if (existing) {
                    existing.remove();
                }
                
                // Create notification
                const notification = document.createElement('div');
                notification.className = 'notification';
                notification.style.background = type === 'success' ? '#10b981' : 
                                              type === 'error' ? '#ef4444' : 
                                              type === 'warning' ? '#f59e0b' : '#3b82f6';
                notification.innerHTML = `
                    <i class="fas fa-${type === 'success' ? 'check-circle' : 
                                      type === 'error' ? 'exclamation-circle' : 
                                      type === 'warning' ? 'exclamation-triangle' : 'info-circle'}"></i>
                    <span>${message}</span>
                `;
                
                document.body.appendChild(notification);
                
                // Remove after 3 seconds
                setTimeout(() => {
                    if (notification.parentNode) {
                        notification.style.animation = 'slideOutRight 0.3s ease-out';
                        setTimeout(() => {
                            if (notification.parentNode) {
                                notification.parentNode.removeChild(notification);
                            }
                        }, 300);
                    }
                }, 3000);
            }

        class ProfileSystem {

                this.auth = authSystem;

                console.log('Loading profile data...');
                this.loadProfileData();

            loadProfileData() {
                const user = this.auth.getCurrentUser();
                console.log('Loading profile for user:', user);
                
                if (!user) {
                    console.log('No user found, redirecting to home');
                    return;
                }
                
                // Update basic profile info
                document.getElementById('userName').textContent = user.name || 'Environmental Hero';
                document.getElementById('userEmail').textContent = user.email || 'user@example.com';
                document.getElementById('userLevel').textContent = `Level ${user.level || 1} Eco-Warrior`;
                
                // Update avatar
                const initial = user.name ? user.name.charAt(0).toUpperCase() : 'U';
                document.getElementById('profileAvatar').innerHTML = initial;
                
                // Update stats
                this.updateStats(user);
                
                // Load learning progress
                this.loadLearningProgress();
                
                // Load badges
                this.loadBadges();
                
                // Load recent activity
                this.loadRecentActivity();
            }
            
            updateStats(user) {
                console.log('Updating stats for user:', user);
                const progress = this.getProgress();
                const topics = ['ecology', 'climate', 'living', 'waste', 'trade', 'policy'];
                
                // Calculate total progress
                let totalProgress = 0;
                topics.forEach(topic => {
                    totalProgress += progress[topic] || 0;
                });
                const avgProgress = Math.round(totalProgress / topics.length);
                
                // Calculate completed quests
                let completedQuests = 0;
                topics.forEach(topic => {
                    completedQuests += Math.floor((progress[topic] || 0) / 20);
                });
                
                // Calculate eco points (10 points per % progress)
                const ecoPoints = Math.round(avgProgress * 10);
                
                // Calculate carbon saved (2kg per % progress)
                const carbonSaved = Math.round(avgProgress * 2);
                
                console.log('Calculated stats:', {
                    completedQuests,
                    ecoPoints,
                    carbonSaved,
                    avgProgress
                });
                
                // Update DOM
                document.getElementById('questsCompleted').textContent = completedQuests;
                document.getElementById('ecoPoints').textContent = ecoPoints.toLocaleString();
                document.getElementById('streakDays').textContent = user.streakDays || 1;
                document.getElementById('carbonSaved').textContent = carbonSaved;
                
                // Update user object
                user.completedQuests = completedQuests;
                user.points = ecoPoints;
                user.carbonSaved = carbonSaved;
                
                // Update level based on points
                const newLevel = Math.floor(ecoPoints / 100) + 1;
                if (newLevel !== user.level) {
                    user.level = newLevel;
                    document.getElementById('userLevel').textContent = `Level ${newLevel} Eco-Warrior`;
                    this.auth.saveCurrentUser(user);
                }
            }
            
            getProgress() {
                try {
                    const progress = localStorage.getItem('greenquest_progress');
                    console.log('Raw progress from localStorage:', progress);
                    
                    if (!progress) {
                        console.log('No progress found, creating default');
                        return {
                            ecology: 0, 
                            climate: 0, 
                            living: 0, 
                            waste: 0, 
                            trade: 0, 
                            policy: 0
                        };
                    }
                    
                    const parsed = JSON.parse(progress);
                    console.log('Parsed progress:', parsed);
                    return parsed;
                    
                } catch (e) {
                    console.error('Error parsing progress:', e);
                    return { 
                        ecology: 0, 
                        climate: 0, 
                        living: 0, 
                        waste: 0, 
                        trade: 0, 
                        policy: 0 
                    };
                }
            }

            loadBadges() {
                console.log('Loading badges...');
                const progress = this.getProgress();
                const user = this.auth.getCurrentUser();
                const badges = [];
                
                // Progress-based badges
                const totalProgress = Object.values(progress).reduce((a, b) => a + b, 0);
                const avgProgress = totalProgress / 6;
                
                console.log('Average progress for badges:', avgProgress);
                
                if (avgProgress >= 20) {
                    badges.push({
                        icon: '🌱',
                        name: 'Green Sprout',
                        desc: 'Completed 20% of learning path'
                    });
                }
                
                if (avgProgress >= 50) {
                    badges.push({
                        icon: '🌿',
                        name: 'Eco Explorer',
                        desc: 'Completed 50% of learning path'
                    });
                }
                
                if (avgProgress >= 80) {
                    badges.push({
                        icon: '🌳',
                        name: 'Forest Guardian',
                        desc: 'Completed 80% of learning path'
                    });
                }
                
                if (avgProgress === 100) {
                    badges.push({
                        icon: '🏆',
                        name: 'Earth Champion',
                        desc: 'Completed entire learning path'
                    });
                }
                
                // Streak badges
                if (user.streakDays >= 7) {
                    badges.push({
                        icon: '🔥',
                        name: 'Weekly Warrior',
                        desc: '7-day streak'
                    });
                }
                
                if (user.streakDays >= 30) {
                    badges.push({
                        icon: '⭐',
                        name: 'Monthly Master',
                        desc: '30-day streak'
                    });
                }
                
                // Always show some starter badges
                badges.push({
                    icon: '🎯',
                    name: 'First Quest',
                    desc: 'Completed your first quest'
                });
                
                badges.push({
                    icon: '🤝',
                    name: 'Community Helper',
                    desc: 'Joined the community'
                });
                
                const container = document.getElementById('badgesGrid');
                if (!container) {
                    console.error('Badges grid container not found!');
                    return;
                }
                
                if (badges.length === 0) {
                    container.innerHTML = '<p style="grid-column: 1 / -1; text-align: center; color: var(--gray-color);">No badges earned yet. Start learning to earn your first badge!</p>';
                    return;
                }
                
                container.innerHTML = badges.map(badge => `
                    <div class="badge-item">
                        <div class="badge-icon">${badge.icon}</div>
                        <div class="badge-name">${badge.name}</div>
                        <div class="badge-desc">${badge.desc}</div>
                    </div>
                `).join('');
                
                console.log('Badges loaded:', badges.length);
            }
