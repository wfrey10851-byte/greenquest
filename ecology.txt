        // 左侧Unit子目录展开/隐藏
        function toggleSubList(subUnitId) {
            const subList = document.getElementById(subUnitId);
            subList.classList.toggle('show');
            // 滚动到对应Unit
            const unitId = subUnitId.replace('sub', '');
            const unitSection = document.getElementById(unitId);
            if (unitSection) {
                unitSection.scrollIntoView({ behavior: 'smooth' });
                // 显示对应Unit，隐藏其他Unit
                document.querySelectorAll('.lesson-section').forEach(section => {
                    section.style.display = 'none';
                });
                unitSection.style.display = 'block';
                // 激活对应Unit链接
                document.querySelectorAll('.unit-link').forEach(link => {
                    link.classList.remove('active');
                });
                document.querySelector(`.unit-link[onclick*="${unitId}"]`).classList.add('active');
            }
        }

        // 设置左侧Unit链接激活状态
        function setActiveUnitLink(link, unitId) {
            // 清除所有Unit链接激活状态
            document.querySelectorAll('.unit-link').forEach(l => l.classList.remove('active'));
            link.classList.add('active');
            
            // 显示对应Unit，隐藏其他
            document.querySelectorAll('.lesson-section').forEach(section => {
                section.style.display = 'none';
            });
            const targetUnit = document.getElementById(unitId);
            if (targetUnit) {
                targetUnit.style.display = 'block';
                targetUnit.scrollIntoView({ behavior: 'smooth', block: 'start' });
                
                // 展开对应Unit的子目录
                const subListId = `sub${unitId}`;
                const subList = document.getElementById(subListId);
                if (subList) {
                    subList.classList.add('show');
                }
            }
        }

        // 登录弹窗逻辑
        const loginBtn = document.getElementById('loginBtn');
        const loginModal = document.getElementById('loginModal');
        const closeLogin = document.getElementById('closeLogin');

        loginBtn.addEventListener('click', () => {
            loginModal.style.display = 'flex';
            document.getElementById('loginEmail').focus();
        });

        closeLogin.addEventListener('click', () => {
            loginModal.style.display = 'none';
        });

        window.addEventListener('click', (e) => {
            if (e.target === loginModal) {
                loginModal.style.display = 'none';
            }
        });

        // 登录表单提交
        document.getElementById('loginForm').addEventListener('submit', (e) => {
            e.preventDefault();
            loginModal.style.display = 'none';
            loginBtn.style.display = 'none';
            document.getElementById('userAvatar').style.display = 'flex';
            showNotification('Login successful! Welcome to GreenQuest', 'success');
        });

        // 通知提示函数
        function showNotification(message, type = 'success') {
            const existing = document.querySelector('.notification');
            if (existing) existing.remove();
            const notification = document.createElement('div');
            notification.className = `notification ${type}`;
            let icon = 'fas fa-check-circle';
            if (type === 'error') icon = 'fas fa-exclamation-circle';
            if (type === 'info') icon = 'fas fa-info-circle';
            notification.innerHTML = `<i class="${icon}"></i><span>${message}</span>`;
            document.body.appendChild(notification);
            setTimeout(() => notification.classList.add('show'), 10);
            setTimeout(() => {
                notification.classList.remove('show');
                setTimeout(() => notification.remove(), 300);
            }, 3000);
        }

        // 页面加载时默认显示Unit1
        window.onload = function() {
            document.getElementById('unit1').style.display = 'block';
        }
    </script>
